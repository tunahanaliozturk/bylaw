# M5: evaluation set and runner, implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** a committed set of 50 labelled pull requests and a `bylaw eval` command that reviews them with any model or host, scores the result, and writes a report that README numbers can quote.

**Architecture:** each case is a small base project (`eval/projects/<name>/`) plus the files a pull request changes (`eval/cases/<id>/head/`) and a `case.toml` with the labels. `bylaw eval` builds a throwaway git repository per case, runs the normal `build_context` on it, sends the brief to a reviewer (an API model, or answers a host model wrote after `bylaw eval prepare`), validates the result exactly as the Action does, and saves one JSON file per case. `bylaw eval score` compares those runs with the labels. Nothing here calls a model unless the user runs `bylaw eval run --provider ...` with a key.

**Tech Stack:** as M3. Standard library `tomllib`, `statistics`, `shutil`, `subprocess`. No new dependencies.

**Spec:** `docs/design.md`, "Testing and evaluation": 40 to 50 labelled pull requests with planted defects (SQL injection, hardcoded secret, path traversal, `.editorconfig` and `CONTRIBUTING` violations, missing acceptance criterion, scope creep) and clean pull requests; reported per model and date: recall and precision per category, false positive rate on clean PRs, share of findings the validator dropped, cost and latency per PR; run before a release, not on every CI run. ADR 0001 (results are per model, not per release).

## Global Constraints

- Everything in the earlier plans' Global Constraints still holds (Python >= 3.13, ruff and mypy strict clean, no new dependencies, English docs with British spelling, no em dashes).
- No model is called by tests or CI. A real model run happens only when the user approves the spend.
- A paid run is never repeated by accident: a run that already saved a result for a case skips it.
- Planted secrets are obviously fake and match no provider's key format, so GitHub secret scanning never blocks a push.
- Files under `eval/` keep their exact bytes (tabs, trailing spaces, missing final newlines are part of the labels): `.gitattributes` marks them `-text`, `eval/.editorconfig` is `root = true` with no rules, and ruff excludes `eval`.
- Code blocks are preceded by `File: <path>` or `Append to: <path>`.

## Review Focus

1. A Windows checkout or an editor rewrites a planted defect (CRLF, trimmed whitespace, added final newline), so a label no longer matches its case. Pinned by the oracle test in Task 6, which runs on the Windows CI legs.
2. A paid run stops halfway (rate limit, network). Re-running must reuse saved results and retry only the failed cases. Test in Task 3.
3. A model reports the same defect twice on neighbouring lines. That is a duplicate, not a false positive. Test in Task 4.
4. A label names a criterion key or rule line that does not exist, so even a perfect reviewer looks wrong. Pinned by the oracle test in Task 6 (validate drops it, the score falls below perfect).
5. A typo in `case.toml` (`catgories`) is ignored and the case silently scores as clean. Loading must fail and name the case. Test in Task 2.

```python
"""A case is a base project plus the files a pull request changes, with labels saying what a review must find."""

import shutil
import subprocess
import tomllib
from pathlib import Path
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, ValidationError

from bylaw.context import build_context
from bylaw.local_git import LocalGitSource
from bylaw.model import Category, CriterionStatus, ReviewContext

Kind = Literal["security", "bug", "rule", "requirement", "scope", "clean"]


class _Frozen(BaseModel):
    model_config = ConfigDict(frozen=True, extra="forbid")


class Span(_Frozen):
    file: str
    lines: tuple[int, int]


class ExpectedFinding(Span):
    categories: tuple[Category, ...]
    note: str
    evidence: str | None = None
    # Found by bylaw's own checks (.editorconfig), with or without a model.
    deterministic: bool = False


class ExpectedCreep(_Frozen):
    file: str
    line: int


class Case(_Frozen):
    id: str
    project: str
    kind: Kind
    title: str
    message: str
    findings: tuple[ExpectedFinding, ...] = ()
    # A legitimate finding here is neither required nor a false positive.
    allowed: tuple[Span, ...] = ()
    criteria: dict[str, tuple[CriterionStatus, ...]] = Field(default_factory=dict)
    scope_creep: tuple[ExpectedCreep, ...] = ()
    remove: tuple[str, ...] = ()


def load_cases(root: Path) -> tuple[Case, ...]:
    cases: list[Case] = []
    for directory in sorted((root / "cases").iterdir()):
        toml = directory / "case.toml"
        if not toml.is_file():
            continue
        try:
            cases.append(Case.model_validate({**tomllib.loads(toml.read_text("utf-8")), "id": directory.name}))
        except (tomllib.TOMLDecodeError, ValidationError) as error:
            raise ValueError(f"eval case {directory.name}: {error}") from error
    return tuple(cases)


def build_case_context(root: Path, case: Case, work: Path) -> ReviewContext:
    """Commit the base project, then the change, and collect the context exactly as `bylaw context --base` does."""
    repo = work / case.id
    shutil.copytree(root / "projects" / case.project, repo)
    _git(repo, "init", "-q", "-b", "main")
    _git(repo, "add", "-A")
    _git(repo, "commit", "-q", "-m", "base")
    _git(repo, "checkout", "-q", "-b", "change")
    head = root / "cases" / case.id / "head"
    if head.is_dir():
        shutil.copytree(head, repo, dirs_exist_ok=True)
    for path in case.remove:
        (repo / path).unlink()
    _git(repo, "add", "-A")
    _git(repo, "commit", "-q", "-m", case.message)
    return build_context(LocalGitSource(repo, "main")).model_copy(update={"title": case.title})


def _git(cwd: Path, *args: str) -> None:
    # autocrlf off: a planted tab, trailing space or missing newline must reach the diff byte for byte.
    settings = ["-c", "user.name=bylaw eval", "-c", "user.email=eval@bylaw.invalid"]
    settings += ["-c", "commit.gpgsign=false", "-c", "core.autocrlf=false"]
    subprocess.run(["git", *settings, *args], cwd=cwd, check=True, capture_output=True)  # noqa: S603, S607
```

```python
"""Review every case with one reviewer and save what would have been posted."""

import json
import time
from collections.abc import Callable, Sequence
from pathlib import Path

from pydantic import BaseModel, ConfigDict, ValidationError

from bylaw.engine import Reviewer, ReviewError, Usage
from bylaw.evaluation.cases import Case, build_case_context
from bylaw.model import ReviewContext, ReviewResult
from bylaw.prompt import render_input, system_prompt
from bylaw.validate import validate


class CaseRun(BaseModel):
    model_config = ConfigDict(frozen=True, extra="forbid")

    case_id: str
    result: ReviewResult
    dropped: tuple[str, ...]
    model_findings: int
    seconds: float | None = None
    input_tokens: int | None = None
    output_tokens: int | None = None
    error: str | None = None


class AnswerReviewer:
    """Replays the result a host model (Claude Code, Codex) wrote after `bylaw eval prepare`."""

    def __init__(self, path: Path) -> None:
        self._path = path

    def review(self, system: str, user: str) -> ReviewResult:  # noqa: ARG002 - the host already read the brief
        if not self._path.is_file():
            raise ReviewError(f"no answer at {self._path.name}")
        raw = self._path.read_bytes()
        try:
            return ReviewResult.model_validate_json(raw.decode(json.detect_encoding(raw)))
        except ValidationError as error:
            raise ReviewError(f"{self._path.name} is not a valid review ({error.error_count()} errors)") from error


class NoModel:
    """bylaw's own checks only: what a repository without a model key gets."""

    def review(self, system: str, user: str) -> ReviewResult:  # noqa: ARG002
        return ReviewResult()


def run_case(context: ReviewContext, case_id: str, reviewer: Reviewer) -> CaseRun:
    text, _ = render_input(context)
    error: str | None = None
    start = time.perf_counter()
    try:
        result = reviewer.review(system_prompt(), text)
    except ReviewError as failure:
        result, error = ReviewResult(), str(failure)
    seconds = time.perf_counter() - start
    merged = result.model_copy(update={"findings": (*context.findings, *result.findings)})
    checked = validate(context, merged)
    # Read only after a success: a reviewer reused across cases still holds the previous case's usage.
    usage: Usage | None = getattr(reviewer, "last_usage", None) if error is None else None
    return CaseRun(
        case_id=case_id,
        result=checked.result,
        dropped=checked.dropped,
        model_findings=len(result.findings),
        seconds=seconds,
        input_tokens=usage.input_tokens if usage else None,
        output_tokens=usage.output_tokens if usage else None,
        error=error,
    )


def run_all(
    root: Path, cases: Sequence[Case], make: Callable[[Case], Reviewer], out: Path, work: Path
) -> tuple[CaseRun, ...]:
    out.mkdir(parents=True, exist_ok=True)
    runs: list[CaseRun] = []
    for case in cases:
        saved = out / f"{case.id}.json"
        if saved.is_file():
            previous = CaseRun.model_validate_json(saved.read_text(encoding="utf-8"))
            # A paid review is never repeated; a failed one is retried.
            if previous.error is None:
                runs.append(previous)
                continue
        run = run_case(build_case_context(root, case, work), case.id, make(case))
        saved.write_text(run.model_dump_json(indent=2), encoding="utf-8")
        runs.append(run)
    return tuple(runs)


def prepare(root: Path, cases: Sequence[Case], out: Path, work: Path) -> None:
    out.mkdir(parents=True, exist_ok=True)
    for case in cases:
        context = build_case_context(root, case, work)
        text, _ = render_input(context)
        (out / f"{case.id}.context.json").write_text(context.model_dump_json(indent=2), encoding="utf-8")
        (out / f"{case.id}.brief.txt").write_text(f"{system_prompt()}\n---\n{text}\n", encoding="utf-8")
```

### Task 4: scoring

**Files:**
- Create: `src/bylaw/evaluation/score.py`
- Test: `tests/unit/test_eval_score.py`

**Interfaces:**
- Consumes: `Case`, `ExpectedFinding`, `ExpectedCreep`, `Span` (Task 2), `CaseRun` (Task 3).
- Produces: `TOLERANCE = 2`; `CategoryScore` (`category`, `true_positives`, `false_positives`, `false_negatives`, properties `precision`/`recall`: `float | None`); `Score` (fields below); `score(cases: Sequence[Case], runs: Sequence[CaseRun]) -> Score`.

Matching rules (these go into `docs/eval/README.md` in Task 7 word for word):
- A finding matches a label when the file is the same, its lines come within `TOLERANCE` lines of the label's lines, and its category is one the label accepts. Each label matches once.
- A second finding that would have matched an already matched label is a duplicate: counted, not a false positive.
- A finding inside an `allowed` span is ignored.
- Any other finding is a false positive in its category. If it sits on an unmatched label with the wrong category, the label counts as located.
- A label nothing matched is a false negative in its first category.
- A clean case "has an inline comment" when a finding of medium severity or above remains after the rules above.
- A criterion is correct when the run's status is one of the accepted statuses.
- Scope creep matches on file and line within `TOLERANCE`.
- A case with no run, or whose run has an error, is listed under errors and scored with what bylaw's own checks found.

```python
"""Compare saved runs with the case labels. The matching rules are listed in docs/eval/README.md."""

import statistics
from collections import Counter
from collections.abc import Sequence

from pydantic import BaseModel, ConfigDict

from bylaw.evaluation.cases import Case, Span
from bylaw.evaluation.run import CaseRun
from bylaw.model import Category, Finding, ReviewResult, Severity

TOLERANCE = 2
_INLINE = frozenset({Severity.CRITICAL, Severity.HIGH, Severity.MEDIUM})


class _Frozen(BaseModel):
    model_config = ConfigDict(frozen=True, extra="forbid")


class CategoryScore(_Frozen):
    category: Category
    true_positives: int
    false_positives: int
    false_negatives: int

    @property
    def precision(self) -> float | None:
        predicted = self.true_positives + self.false_positives
        return self.true_positives / predicted if predicted else None

    @property
    def recall(self) -> float | None:
        labelled = self.true_positives + self.false_negatives
        return self.true_positives / labelled if labelled else None


class Score(_Frozen):
    cases: int
    categories: tuple[CategoryScore, ...]
    labelled: int
    located: int
    duplicates: int
    clean_cases: int
    clean_with_inline: int
    model_findings: int
    dropped: int
    criteria_total: int
    criteria_correct: int
    creep_expected: int
    creep_found: int
    creep_extra: int
    median_seconds: float | None
    p90_seconds: float | None
    input_tokens: int | None
    output_tokens: int | None
    errors: tuple[str, ...]
    misses: tuple[str, ...]
    false_alarms: tuple[str, ...]


def _near(finding: Finding, span: Span) -> bool:
    start, end = span.lines
    return (
        finding.file == span.file
        and finding.line_start <= end + TOLERANCE
        and finding.line_end >= start - TOLERANCE
    )


def score(cases: Sequence[Case], runs: Sequence[CaseRun]) -> Score:  # noqa: PLR0912, PLR0915 - one pass, many counters
    by_id = {run.case_id: run for run in runs}
    tp: Counter[Category] = Counter()
    fp: Counter[Category] = Counter()
    fn: Counter[Category] = Counter()
    located = duplicates = clean_cases = clean_with_inline = 0
    criteria_total = criteria_correct = creep_expected = creep_found = creep_extra = 0
    errors: list[str] = []
    misses: list[str] = []
    false_alarms: list[str] = []
    for case in cases:
        run = by_id.get(case.id)
        if run is None:
            errors.append(f"{case.id}: no run")
            result = ReviewResult()
        else:
            if run.error is not None:
                errors.append(f"{case.id}: {run.error}")
            result = run.result
        open_labels = list(case.findings)
        matched: list[Span] = []
        found_located: set[int] = set()
        inline = False
        for finding in result.findings:
            hit = next(
                (e for e in open_labels if _near(finding, e) and finding.category in e.categories), None
            )
            if hit is not None:
                open_labels.remove(hit)
                matched.append(hit)
                tp[finding.category] += 1
                located += 1
                continue
            if any(_near(finding, e) for e in matched):
                duplicates += 1
                continue
            if any(_near(finding, span) for span in case.allowed):
                continue
            fp[finding.category] += 1
            false_alarms.append(
                f"{case.id}: {finding.file}:{finding.line_start} {finding.category} {finding.message[:80]}"
            )
            inline = inline or finding.severity in _INLINE
            for label in open_labels:
                if _near(finding, label) and id(label) not in found_located:
                    found_located.add(id(label))
                    located += 1
                    break
        for label in open_labels:
            fn[label.categories[0]] += 1
            misses.append(f"{case.id}: {label.note}")
        if case.kind == "clean":
            clean_cases += 1
            clean_with_inline += inline
        statuses = {c.criterion: c.status for c in result.criteria}
        for key, accepted in case.criteria.items():
            criteria_total += 1
            criteria_correct += statuses.get(key) in accepted
        open_creep = list(case.scope_creep)
        creep_expected += len(open_creep)
        for creep in result.scope_creep:
            hit_creep = next(
                (c for c in open_creep if c.file == creep.file and abs(c.line - creep.line) <= TOLERANCE), None
            )
            if hit_creep is None:
                creep_extra += 1
            else:
                open_creep.remove(hit_creep)
                creep_found += 1
    seconds = sorted(r.seconds for r in runs if r.seconds is not None and r.error is None)
    tokens_in = [r.input_tokens for r in runs if r.input_tokens is not None]
    tokens_out = [r.output_tokens for r in runs if r.output_tokens is not None]
    return Score(
        cases=len(cases),
        categories=tuple(
            CategoryScore(category=c, true_positives=tp[c], false_positives=fp[c], false_negatives=fn[c])
            for c in Category
        ),
        labelled=sum(len(c.findings) for c in cases),
        located=located,
        duplicates=duplicates,
        clean_cases=clean_cases,
        clean_with_inline=clean_with_inline,
        model_findings=sum(r.model_findings for r in runs),
        dropped=sum(len(r.dropped) for r in runs),
        criteria_total=criteria_total,
        criteria_correct=criteria_correct,
        creep_expected=creep_expected,
        creep_found=creep_found,
        creep_extra=creep_extra,
        median_seconds=statistics.median(seconds) if seconds else None,
        p90_seconds=statistics.quantiles(seconds, n=10)[-1] if len(seconds) >= 2 else None,  # noqa: PLR2004
        input_tokens=sum(tokens_in) if tokens_in else None,
        output_tokens=sum(tokens_out) if tokens_out else None,
        errors=tuple(errors),
        misses=tuple(misses),
        false_alarms=tuple(false_alarms),
    )
```

```python
"""The Markdown report README numbers are quoted from."""

from bylaw.evaluation.score import Score


def _percent(value: float | None) -> str:
    return "n/a" if value is None else f"{value:.0%}"


def render(score: Score, label: str, *, input_price: float | None = None, output_price: float | None = None) -> str:
    lines = [
        f"# Evaluation: {label}",
        "",
        f"{score.cases} cases. Matching rules: docs/eval/README.md.",
        "",
        "| Category | Hits | False alarms | Misses | Precision | Recall |",
        "| --- | --- | --- | --- | --- | --- |",
    ]
    lines += [
        f"| {c.category} | {c.true_positives} | {c.false_positives} | {c.false_negatives} | "
        f"{_percent(c.precision)} | {_percent(c.recall)} |"
        for c in score.categories
    ]
    lines += [
        "",
        f"- Labelled defects found at the right place, any category: {score.located} of {score.labelled}.",
        f"- Clean pull requests with an inline comment (medium or above): {score.clean_with_inline} of "
        f"{score.clean_cases}.",
        f"- Model findings the validator dropped: {score.dropped} of {score.model_findings}.",
        f"- Duplicate findings: {score.duplicates}.",
        f"- Acceptance criteria judged correctly: {score.criteria_correct} of {score.criteria_total}.",
        f"- Scope creep found: {score.creep_found} of {score.creep_expected}, with {score.creep_extra} extra.",
    ]
    if score.median_seconds is not None:
        p90 = f", {score.p90_seconds:.1f} s p90" if score.p90_seconds is not None else ""
        lines.append(f"- Latency: {score.median_seconds:.1f} s median{p90}.")
    if score.input_tokens is not None and score.output_tokens is not None:
        lines.append(f"- Tokens: {score.input_tokens:,} in, {score.output_tokens:,} out.")
        if input_price is not None and output_price is not None and score.cases:
            cost = (score.input_tokens * input_price + score.output_tokens * output_price) / 1_000_000
            lines.append(
                f"- Cost: ${cost / score.cases:.2f} per pull request at the given prices, an upper bound "
                "because cached input is billed as full input."
            )
    for title, items in (("Errors", score.errors), ("Misses", score.misses), ("False alarms", score.false_alarms)):
        if items:
            lines += ["", f"## {title}", "", *(f"- {item}" for item in items)]
    return "\n".join(lines) + "\n"
```

```python
"""The committed evaluation set: every label must be reachable, and a reviewer that knows the labels scores perfectly."""

from collections import Counter
from pathlib import Path

import pytest

from bylaw.evaluation.cases import Case, load_cases
from bylaw.evaluation.run import run_all
from bylaw.evaluation.score import score
from bylaw.model import CriterionResult, Finding, ReviewResult, ScopeCreep, Severity

ROOT = Path(__file__).resolve().parents[2] / "eval"


class Oracle:
    """Answers with exactly the labels; whatever validate drops or the scorer misses is a labelling error."""

    def __init__(self, case: Case) -> None:
        self._case = case

    def review(self, system: str, user: str) -> ReviewResult:  # noqa: ARG002
        findings = tuple(
            Finding(
                file=label.file,
                line_start=label.lines[0],
                line_end=label.lines[1],
                category=label.categories[0],
                severity=Severity.HIGH,
                message=label.note,
                evidence=label.evidence,
            )
            for label in self._case.findings
            if not label.deterministic
        )
        criteria = tuple(
            CriterionResult(criterion=key, status=accepted[0]) for key, accepted in self._case.criteria.items()
        )
        creep = tuple(ScopeCreep(file=c.file, line=c.line, message="creep") for c in self._case.scope_creep)
        return ReviewResult(findings=findings, criteria=criteria, scope_creep=creep)


@pytest.fixture(scope="module")
def cases() -> tuple[Case, ...]:
    return load_cases(ROOT)


def test_an_oracle_reviewer_scores_perfectly(cases: tuple[Case, ...], tmp_path_factory: pytest.TempPathFactory) -> None:
    tmp = tmp_path_factory.mktemp("oracle")

    runs = run_all(ROOT, cases, Oracle, tmp / "runs", tmp / "work")
    result = score(cases, runs)

    assert [r.dropped for r in runs if r.dropped] == []
    assert result.misses == ()
    assert result.false_alarms == ()
    assert result.duplicates == 0
    assert result.criteria_correct == result.criteria_total
    assert (result.creep_found, result.creep_extra) == (result.creep_expected, 0)
    assert result.clean_with_inline == 0
```
